cmake_minimum_required(VERSION 3.16)
project(uefiwelcome LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(KF6CoreAddons REQUIRED)
find_package(Calamares REQUIRED)
add_library(uefiwelcome MODULE Welcome.cpp)
target_link_libraries(uefiwelcome PRIVATE Calamares::calamaresui)
target_compile_definitions(uefiwelcome PRIVATE UPSTREAM_WELCOME="${UPSTREAM_WELCOME}")
install(TARGETS uefiwelcome LIBRARY DESTINATION lib/calamares/modules/uefiwelcome)
install(FILES module.desc DESTINATION lib/calamares/modules/uefiwelcome)
